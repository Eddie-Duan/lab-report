% 导言区相对原模板做了裁剪: 只留本机 MiKTeX 里确实装着的包, 免得首次编译就卡在缺包上。
% 原模板还有 algorithm/algpseudocode、mdframed、adjustbox、fontsize、wrapfig、rotating、
% breqn、mathtools、mathrsfs、amsthm、multicol、siunitx、circuitikz、subfigure —— 需要时
% 照 原版参考.tex 的导言区加回来即可 (MiKTeX 会自动联网装; subfigure 已废弃, 用 subcaption)。

\documentclass[11pt]{article}

\usepackage[a4paper]{geometry}
\geometry{left=2.0cm,right=2.0cm,top=2.5cm,bottom=2.5cm}

\usepackage{ctex} % 支持中文的LaTeX宏包
\usepackage{amsmath,amsfonts,amssymb,bm} % 数学公式和符号
\usepackage{graphicx} % 插图
\usepackage{pdfpages} % \includepdf: 把预习报告/数据记录等 PDF 附在报告后面（附录用）
\usepackage{fontspec} % 字体设置
\usepackage{booktabs,multirow,array,bigstrut} % 三线表与表格
\usepackage{float} % 让 [H] 生效(图/表钉在原地)
\usepackage{caption,subcaption} % 图表标题; subcaption 是 subfigure 的现代替代
\usepackage{enumitem} % 列表间距
\usepackage{fancyhdr} % 页眉页脚
\usepackage{listings} % 插入源代码(处理数据的脚本可以贴)
\usepackage{tikz,xcolor} % 画图与颜色
\usepackage[capitalize]{cleveref} % \cref{} 自动加前缀

\definecolor{dkgreen}{rgb}{0,0.6,0}
\definecolor{gray}{rgb}{0.5,0.5,0.5}
\definecolor{mauve}{rgb}{0.58,0,0.82}
\lstset{
  frame=tb,
  aboveskip=3mm,
  belowskip=3mm,
  showstringspaces=false,
  columns=flexible,
  framerule=1pt,
  rulecolor=\color{gray!35},
  backgroundcolor=\color{gray!5},
  basicstyle={\small\ttfamily},
  numbers=none,
  numberstyle=\tiny\color{gray},
  keywordstyle=\color{blue},
  commentstyle=\color{dkgreen},
  stringstyle=\color{mauve},
  breaklines=true,
  breakatwhitespace=true,
  tabsize=3,
}

\Crefname{section}{Section}{Sections}
\Crefname{table}{Table}{Tables}
\crefname{table}{Table.}{Tabs.}

% 字体: 原模板用的是随附的 Palatino Linotype.ttf / SimHei.ttf; 这里改成系统字体名,

\setmainfont{Palatino Linotype}
